# --- 1. VIGIADOR DE CATRACA ---
def monitorar_presenca():
    ultimo_id_lido = 0
    print(f"🚀 Vigiador iniciado na porta {DB_CONFIG['port']}... Monitorando catraca.")
    
    while True:
        try:
            conn = psycopg2.connect(**DB_CONFIG)
            cur = conn.cursor()
            
            # Busca o último evento (aluno passando na catraca)
            cur.execute("SELECT id, pessoa_id, data_hora FROM evento_dispositivo ORDER BY id DESC LIMIT 1")
            evento = cur.fetchone()
            
            if evento:
                idx, aluno_id, timestamp = evento
                
                # Se for um registro novo
                if ultimo_id_lido != 0 and idx > ultimo_id_lido:
                    print(f"📢 Entrada detectada! Aluno ID: {aluno_id}")
                    
                    payload = {
                        "aluno_id": str(aluno_id), 
                        "timestamp": str(timestamp)
                    }
                    
                    try:
                        res = requests.post(WEBHOOK_REPLIT, json=payload, timeout=10)
                        print(f"✅ Sincronizado com Replit: {res.status_code}")
                    except Exception as e:
                        print(f"⚠️ Erro ao enviar para nuvem: {e}")
                
                ultimo_id_lido = idx
            
            cur.close()
            conn.close()
            
        except Exception:
            # Mensagem silenciosa para não poluir o terminal se estiver tentando reconectar
            pass
        
        time.sleep(2)
