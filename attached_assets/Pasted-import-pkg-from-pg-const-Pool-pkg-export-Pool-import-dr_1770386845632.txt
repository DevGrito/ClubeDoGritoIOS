import pkg from "pg";
const {Pool} = pkg;
export {Pool};
import { drizzle, NodePgDatabase } from "drizzle-orm/node-postgres";
import * as schema from "@shared/schema";

type DbConfig = {
  host: string;
  port: number;
  user: string;
  password: string;
  database: string;
  ssl: boolean | { rejectUnauthorized: boolean };
  source: string;
};

function getDigitalOceanConfig(): DbConfig | null {
  if (process.env.DO_DATABASE_URL) {
    const url = new URL(process.env.DO_DATABASE_URL);
    // Forçar porta 5433 (porta pública do container Docker)
    const port = parseInt(process.env.DO_DB_PORT || '5433');
    return {
      host: url.hostname,
      port: port,
      user: decodeURIComponent(url.username),
      password: decodeURIComponent(url.password),
      database: url.pathname.slice(1),
      ssl: false,
      source: "Digital Ocean"
    };
  }
  if (process.env.DO_DB_HOST && process.env.DO_DB_USER && process.env.DO_DB_PASSWORD && process.env.DO_DB_NAME) {
    return {
      host: process.env.DO_DB_HOST,
      port: parseInt(process.env.DO_DB_PORT || '5433'),
      user: process.env.DO_DB_USER,
      password: process.env.DO_DB_PASSWORD,
      database: process.env.DO_DB_NAME,
      ssl: false,
      source: "Digital Ocean"
    };
  }
  return null;
}

function getNeonConfig(): DbConfig | null {
  if (process.env.DATABASE_URL) {
    const url = new URL(process.env.DATABASE_URL);
    return {
      host: url.hostname,
      port: parseInt(url.port || '5432'),
      user: decodeURIComponent(url.username),
      password: decodeURIComponent(url.password),
      database: url.pathname.slice(1),
      ssl: { rejectUnauthorized: false },
      source: "Neon"
    };
  }
  return null;
}

const doConfig = getDigitalOceanConfig();
const neonConfig = getNeonConfig();

if (!doConfig && !neonConfig) {
  throw new Error("❌ ERRO CRÍTICO: Nenhum banco de dados configurado!");
}

let activeConfig = doConfig || neonConfig!;
let dbSource = activeConfig.source;

console.log(`🔌 CONECTANDO AO BANCO: ${dbSource} (${activeConfig.host}:${activeConfig.port}/${activeConfig.database})`);

const poolConfig = {
  host: activeConfig.host,
  port: activeConfig.port,
  user: activeConfig.user,
  password: activeConfig.password,
  database: activeConfig.database,
  ssl: activeConfig.ssl,
  max: 20,
  idleTimeoutMillis: 30_000,
  connectionTimeoutMillis: 10_000,
  options: '-c search_path=public',
};

export let pool = new Pool(poolConfig);

pool.on('connect', async (client) => {
  try {
    await client.query('SET search_path TO public');
  } catch (err) {
    console.error('❌ Erro ao definir search_path:', err);
  }
});

pool.on("error", (err) => {
  console.error("❌ Database pool error:", err);
});

async function tryConnect(config: DbConfig, retries: number = 3): Promise<boolean> {
  const testPool = new Pool({
    host: config.host,
    port: config.port,
    user: config.user,
    password: config.password,
    database: config.database,
    ssl: config.ssl,
    connectionTimeoutMillis: 5000,
  });

  for (let attempt = 1; attempt <= retries; attempt++) {
    try {
      console.log(`🔄 [${config.source}] Tentativa ${attempt}/${retries}...`);
      const client = await testPool.connect();
      const result = await client.query("SELECT current_database() as db_name");
      client.release();
      await testPool.end();
      console.log(`✅ ${config.source} conectado: ${result.rows[0].db_name}`);
      return true;
    } catch (error) {
      console.error(`❌ [${config.source}] Tentativa ${attempt} falhou:`, error instanceof Error ? error.message : error);
      if (attempt < retries) {
        await new Promise(resolve => setTimeout(resolve, 2000));
      }
    }
  }
  await testPool.end().catch(() => {});
  return false;
}

export async function testDatabaseConnection(): Promise<boolean> {
  if (doConfig) {
    const doConnected = await tryConnect(doConfig);
    if (doConnected) {
      dbSource = "Digital Ocean";
      return true;
    }
    console.log("⚠️ Digital Ocean inacessível, tentando Neon como fallback...");
  }

  if (neonConfig) {
    const neonConnected = await tryConnect(neonConfig);
    if (neonConnected) {
      console.log("🔄 Reconfigurando pool para Neon...");
      await pool.end().catch(() => {});

      pool = new Pool({
        host: neonConfig.host,
        port: neonConfig.port,
        user: neonConfig.user,
        password: neonConfig.password,
        database: neonConfig.database,
        ssl: neonConfig.ssl,
        max: 20,
        idleTimeoutMillis: 30_000,
        connectionTimeoutMillis: 10_000,
        options: '-c search_path=public',
      });

      pool.on('connect', async (client) => {
        try {
          await client.query('SET search_path TO public');
        } catch (err) {
          console.error('❌ Erro ao definir search_path:', err);
        }
      });

      pool.on("error", (err) => {
        console.error("❌ Neon pool error:", err);
      });

      db = drizzle(pool, { schema, logger: true });

      dbSource = "Neon (fallback)";
      return true;
    }
  }

  throw new Error("❌ Nenhum banco de dados disponível!");
}

export let db: NodePgDatabase<typeof schema> = drizzle(pool, { 
  schema,
  logger: {
    logQuery(query, params) {}
  }
});

export function refreshDrizzle() {
  db = drizzle(pool, { schema, logger: true });
}

// Migração automática para adicionar colunas novas
export async function runAutoMigrations() {
  try {
    console.log('🔄 Verificando migrações pendentes...');

    // Adicionar colunas de horário ao monitor_grupos se não existirem
    await pool.query(`
      ALTER TABLE monitor_grupos 
      ADD COLUMN IF NOT EXISTS horario_inicio TEXT,
      ADD COLUMN IF NOT EXISTS horario_fim TEXT,
      ADD COLUMN IF NOT EXISTS dias_semana TEXT[];
    `);

    console.log('✅ Migrações automáticas concluídas');
  } catch (error) {
    console.error('⚠️ Erro nas migrações automáticas (pode ser ignorado se colunas já existem):', error instanceof Error ? error.message : error);
  }
}
