from flask import Flask, request, jsonify
import sqlite3 # Ou PostgreSQL se estiver usando no Replit
from datetime import datetime, timedelta

app = Flask(__name__)

# 1. Configuração do Banco de Dados do Replit
def init_db():
    conn = sqlite3.connect('gestao_instituto.db')
    cursor = conn.cursor()
    
    # Tabela de Aulas (A base de tudo)
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS aulas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            turma_id TEXT,
            nome_curso TEXT,
            data_aula DATE,
            hora_inicio TIME,
            hora_fim TIME
        )
    ''')
    
    # Nova Tabela de Presenças com Regra Anti-Duplicidade
    # UNIQUE(aula_id, aluno_id) garante que o João não tenha 2 presenças na MESMA aula
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS presencas_novas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            aula_id INTEGER,
            aluno_id TEXT,
            nome_aluno TEXT,
            horario_entrada TIMESTAMP,
            unidade TEXT,
            UNIQUE(aula_id, aluno_id), 
            FOREIGN KEY (aula_id) REFERENCES aulas (id)
        )
    ''')
    conn.commit()
    conn.close()

init_db()

# 2. Rota do Webhook (Onde o seu PC envia os dados)
@app.route('/webhook', methods=['POST'])
def receber_giro():
    dados = request.json
    if not dados:
        return jsonify({"erro": "Sem dados"}), 400

    aluno_id = dados.get('aluno_id')
    nome = dados.get('nome')
    # O timestamp que vem do seu PC (ex: 2026-03-04 11:30:00)
    horario_giro = datetime.strptime(dados.get('horario'), '%Y-%m-%d %H:%M:%S')
    data_hoje = horario_giro.date()
    hora_atual = horario_giro.strftime('%H:%M:%S')

    conn = sqlite3.connect('gestao_instituto.db')
    cursor = conn.cursor()

    try:
        # 🧠 A LÓGICA DO MATCH: Busca uma aula para este aluno agora
        # Consideramos 15 min de tolerância antes e depois do início
        cursor.execute('''
            SELECT id, nome_curso FROM aulas 
            WHERE data_aula = ? 
            AND ? BETWEEN time(hora_inicio, '-20 minutes') AND time(hora_inicio, '+45 minutes')
        ''', (data_hoje, hora_atual))
        
        aula = cursor.fetchone()

        if aula:
            aula_id, nome_curso = aula
            # Tenta inserir a presença. Se já existir (UNIQUE), o banco barra.
            cursor.execute('''
                INSERT INTO presencas_novas (aula_id, aluno_id, nome_aluno, horario_entrada, unidade)
                VALUES (?, ?, ?, ?, ?)
            ''', (aula_id, aluno_id, nome, horario_giro, dados.get('unidade')))
            
            conn.commit()
            print(f"✅ Presença confirmada: {nome} na aula de {nome_curso}")
            return jsonify({"status": "sucesso", "msg": f"Presença em {nome_curso}"}), 200
        else:
            print(f"⚠️ {nome} passou, mas não há aula iniciada agora.")
            return jsonify({"status": "ignorado", "msg": "Sem aula no horário"}), 200

    except sqlite3.IntegrityError:
        print(f"🚫 Duplicidade barrada: {nome} já tem presença nesta aula.")
        return jsonify({"status": "duplicado", "msg": "Já registrado"}), 200
    finally:
        conn.close()

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
